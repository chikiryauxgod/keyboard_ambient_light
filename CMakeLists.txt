cmake_minimum_required(VERSION 3.20)
project(KeyboardAmbient VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AMBIENT_BUILD_GUI "Build the Qt desktop application" ON)
option(AMBIENT_REQUIRE_PORTAL "Require Qt Multimedia capture (Qt 6.5+)" OFF)
option(AMBIENT_BUNDLE_RUNTIME "Deploy Qt runtime alongside installed application" OFF)
option(AMBIENT_REQUIRE_HIDAPI "Fail rather than build without VIA support" OFF)
option(AMBIENT_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(AMBIENT_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(AMBIENT_ENABLE_COVERAGE "Enable GCC coverage instrumentation" OFF)
set(AMBIENT_RELEASE_TAG "${PROJECT_VERSION}" CACHE STRING "Release tag shown by the application")
if(NOT AMBIENT_RELEASE_TAG MATCHES "^[A-Za-z0-9][A-Za-z0-9._/-]*$")
  message(FATAL_ERROR "Invalid AMBIENT_RELEASE_TAG")
endif()
add_library(ambient_options INTERFACE)
if(MSVC)
  target_compile_options(ambient_options INTERFACE /W4 /utf-8)
  if(AMBIENT_WARNINGS_AS_ERRORS)
    target_compile_options(ambient_options INTERFACE /WX)
  endif()
else()
  target_compile_options(ambient_options INTERFACE -Wall -Wextra -Wpedantic)
  if(AMBIENT_WARNINGS_AS_ERRORS)
    target_compile_options(ambient_options INTERFACE -Werror)
  endif()
endif()
if(AMBIENT_ENABLE_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Combined ASan/UBSan configuration requires GCC or Clang")
  endif()
  target_compile_options(ambient_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ambient_options INTERFACE -fsanitize=address,undefined)
endif()
if(AMBIENT_ENABLE_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "Coverage configuration requires GCC")
  endif()
  target_compile_options(ambient_options INTERFACE --coverage -O0 -g)
  target_link_options(ambient_options INTERFACE --coverage)
endif()
include(CTest)
add_library(ambient_core src/core.cpp)
target_include_directories(ambient_core PUBLIC include)
target_link_libraries(ambient_core PUBLIC ambient_options)
if(BUILD_TESTING)
  find_package(GTest 1.17 CONFIG QUIET)
  if(NOT GTest_FOUND)
    include(FetchContent)
    if(POLICY CMP0135)
      cmake_policy(SET CMP0135 NEW)
    endif()
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest
      URL https://github.com/google/googletest/archive/refs/tags/v1.18.0.tar.gz
      URL_HASH SHA256=6e3191c1455468b3fc35a417fb565c1c5071aee1b7e7f85e30cf48a98d37d8b5)
    FetchContent_MakeAvailable(googletest)
  endif()
  include(GoogleTest)
  add_executable(ambient_tests tests/core_test.cpp)
  target_link_libraries(ambient_tests PRIVATE ambient_core GTest::gtest_main)
  gtest_discover_tests(ambient_tests DISCOVERY_MODE PRE_TEST NO_PRETTY_VALUES)
  find_package(Python3 3.11 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
    add_test(NAME release_tools COMMAND ${Python3_EXECUTABLE} -W error::ResourceWarning -m unittest discover -s tests -p test_release_tools.py -v)
    set_tests_properties(release_tools PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
endif()
if(AMBIENT_BUILD_GUI)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network)
  set(CMAKE_AUTOMOC ON)
  if(Qt6_VERSION VERSION_GREATER_EQUAL "6.5")
    find_package(Qt6 QUIET COMPONENTS Multimedia)
  endif()
  if(AMBIENT_REQUIRE_PORTAL AND NOT TARGET Qt6::Multimedia)
    message(FATAL_ERROR "Qt 6.5+ Multimedia required for portal/streaming capture")
  endif()
  find_package(PkgConfig QUIET)
  find_package(hidapi CONFIG QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(HIDAPI QUIET IMPORTED_TARGET hidapi-hidraw)
  endif()
  if(NOT TARGET PkgConfig::HIDAPI)
    find_path(HIDAPI_INCLUDE_DIR hidapi/hidapi.h)
    find_library(HIDAPI_LIBRARY NAMES hidapi hidapi-hidraw)
  endif()
  add_library(ambient_devices src/backends.cpp)
  target_link_libraries(ambient_devices PUBLIC ambient_core Qt6::Network)
  target_include_directories(ambient_devices PUBLIC src)
  add_library(ambient_capture src/capture.cpp src/capture.hpp)
  target_include_directories(ambient_capture PUBLIC src)
  target_link_libraries(ambient_capture PUBLIC Qt6::Gui ambient_options)
  if(TARGET Qt6::Multimedia)
    target_link_libraries(ambient_capture PRIVATE Qt6::Multimedia)
    target_compile_definitions(ambient_capture PRIVATE AMBIENT_MULTIMEDIA)
  endif()
  qt_add_executable(keyboard-ambient src/main.cpp)
  target_link_libraries(keyboard-ambient PRIVATE ambient_capture)
  target_link_libraries(keyboard-ambient PRIVATE ambient_devices Qt6::Widgets)
  target_include_directories(keyboard-ambient PRIVATE src)
  target_compile_definitions(keyboard-ambient PRIVATE AMBIENT_RELEASE_TAG="${AMBIENT_RELEASE_TAG}")
  if(TARGET hidapi::hidapi)
    target_link_libraries(ambient_devices PRIVATE hidapi::hidapi)
    target_compile_definitions(ambient_devices PRIVATE AMBIENT_HIDAPI)
  elseif(TARGET PkgConfig::HIDAPI)
    target_link_libraries(ambient_devices PRIVATE PkgConfig::HIDAPI)
    target_compile_definitions(ambient_devices PRIVATE AMBIENT_HIDAPI)
  elseif(HIDAPI_INCLUDE_DIR AND HIDAPI_LIBRARY)
    target_include_directories(ambient_devices PRIVATE ${HIDAPI_INCLUDE_DIR})
    target_link_libraries(ambient_devices PRIVATE ${HIDAPI_LIBRARY})
    target_compile_definitions(ambient_devices PRIVATE AMBIENT_HIDAPI)
  else()
    if(AMBIENT_REQUIRE_HIDAPI)
      message(FATAL_ERROR "HIDAPI is required for this build")
    endif()
    message(WARNING "HIDAPI missing: preview and OpenRGB available; VIA disabled")
  endif()
  if(BUILD_TESTING)
    add_executable(ambient_backend_tests tests/backend_test.cpp tests/qt_test_main.cpp)
    target_link_libraries(ambient_backend_tests PRIVATE ambient_devices GTest::gtest)
    gtest_discover_tests(ambient_backend_tests DISCOVERY_MODE PRE_TEST NO_PRETTY_VALUES)
    add_executable(ambient_capture_tests tests/capture_test.cpp tests/qt_test_main.cpp)
    target_link_libraries(ambient_capture_tests PRIVATE ambient_capture GTest::gtest)
    gtest_discover_tests(ambient_capture_tests DISCOVERY_MODE PRE_TEST NO_PRETTY_VALUES)
    add_executable(ambient_via_tests tests/via_test.cpp tests/qt_test_main.cpp src/backends.cpp)
    target_include_directories(ambient_via_tests PRIVATE tests/fakes src)
    target_compile_definitions(ambient_via_tests PRIVATE AMBIENT_HIDAPI AMBIENT_FAKE_HIDAPI)
    target_link_libraries(ambient_via_tests PRIVATE ambient_core Qt6::Network GTest::gtest)
    gtest_discover_tests(ambient_via_tests DISCOVERY_MODE PRE_TEST NO_PRETTY_VALUES)
    add_test(NAME ambient_gui_startup COMMAND keyboard-ambient --smoke-test)
    set_tests_properties(ambient_gui_startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
    if(AMBIENT_ENABLE_SANITIZERS)
      # Qt's offscreen platform retains process-global allocations outside our control.
      set_property(TEST ambient_gui_startup APPEND PROPERTY ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
    endif()
  endif()
  install(TARGETS keyboard-ambient RUNTIME DESTINATION bin BUNDLE DESTINATION .)
  if(UNIX AND NOT APPLE)
    install(FILES packaging/org.keyboardambient.desktop DESTINATION share/applications)
  endif()
  if(APPLE)
    set_target_properties(keyboard-ambient PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_GUI_IDENTIFIER "org.keyboardambient.desktop"
      MACOSX_BUNDLE_BUNDLE_NAME "Keyboard Ambient")
  elseif(WIN32)
    set_target_properties(keyboard-ambient PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  if(AMBIENT_BUNDLE_RUNTIME)
    if(Qt6_VERSION VERSION_LESS "6.5")
      message(FATAL_ERROR "Runtime deployment requires Qt 6.5+; use Qt 6.8 for release builds")
    endif()
    if(UNIX AND NOT APPLE)
      set_target_properties(keyboard-ambient PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    endif()
    qt_generate_deploy_app_script(TARGET keyboard-ambient OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${deploy_script})
    if(UNIX AND NOT APPLE)
      # Qt's generic deployment may select only XCB on an X11 build host.
      # Include Wayland explicitly so the release is not tied to the build session.
      get_target_property(AMBIENT_QT_CORE Qt6::Core IMPORTED_LOCATION_RELEASE)
      if(NOT AMBIENT_QT_CORE)
        get_target_property(AMBIENT_QT_CORE Qt6::Core IMPORTED_LOCATION)
      endif()
      get_filename_component(AMBIENT_QT_LIB_DIR "${AMBIENT_QT_CORE}" DIRECTORY)
      set(AMBIENT_QT_PLUGIN_DIR "${AMBIENT_QT_LIB_DIR}/../plugins")
      foreach(category wayland-graphics-integration-client wayland-shell-integration wayland-decoration-client)
        if(EXISTS "${AMBIENT_QT_PLUGIN_DIR}/${category}")
          install(DIRECTORY "${AMBIENT_QT_PLUGIN_DIR}/${category}" DESTINATION plugins)
        endif()
      endforeach()
      file(GLOB AMBIENT_EXTRA_PLATFORMS "${AMBIENT_QT_PLUGIN_DIR}/platforms/*wayland*.so" "${AMBIENT_QT_PLUGIN_DIR}/platforms/*offscreen*.so")
      file(GLOB AMBIENT_WAYLAND_LIBRARIES "${AMBIENT_QT_LIB_DIR}/libQt6Wayland*.so*")
      install(FILES ${AMBIENT_EXTRA_PLATFORMS} DESTINATION plugins/platforms)
      install(FILES ${AMBIENT_WAYLAND_LIBRARIES} DESTINATION lib)
      find_library(AMBIENT_HID_RUNTIME NAMES hidapi-hidraw hidapi)
      if(AMBIENT_HID_RUNTIME)
        get_filename_component(AMBIENT_HID_DIR "${AMBIENT_HID_RUNTIME}" DIRECTORY)
        file(GLOB AMBIENT_HID_LIBRARIES "${AMBIENT_HID_DIR}/libhidapi-hidraw.so.*" "${AMBIENT_HID_DIR}/libhidapi.so.*")
        install(FILES ${AMBIENT_HID_LIBRARIES} DESTINATION lib)
      endif()
    endif()
  endif()
  if(WIN32 AND AMBIENT_BUNDLE_RUNTIME)
    find_file(AMBIENT_HID_DLL hidapi.dll HINTS
      "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/bin"
      "${HIDAPI_INCLUDE_DIR}/../bin")
    if(AMBIENT_REQUIRE_HIDAPI AND NOT AMBIENT_HID_DLL)
      message(FATAL_ERROR "HIDAPI runtime DLL required for Windows release package")
    endif()
    if(AMBIENT_HID_DLL)
      install(FILES ${AMBIENT_HID_DLL} DESTINATION bin)
    endif()
  endif()
endif()
set(CPACK_PACKAGE_NAME "KeyboardAmbient")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "${CPACK_PACKAGE_NAME}-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_PACKAGE_VENDOR "Keyboard Ambient")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Screen-driven keyboard ambient lighting")
set(CPACK_GENERATOR "ZIP")
if(UNIX AND NOT APPLE)
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
